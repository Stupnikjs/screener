===== cache.py =====
"""
Cache disque persistant pour le scanner : cours de clôture, PER et ROIC.

Disposition (dans le dossier racine, par défaut `.cache` à côté du CSV) :
    prices/px_<ticker>.csv   un fichier par titre : colonnes date, close (cours ajustés)
    prices_meta.json         par titre : date du dernier téléchargement, début couvert, statut
    pe.json                  PER (12 mois / prévu) avec date de téléchargement
    roic.json                ROIC (dernier exercice / moyenne) avec date de téléchargement

Principes :
- un titre « à jour » (téléchargé depuis moins de `ttl_hours`) ne génère aucune requête ;
- un titre plus ancien est complété : on ne retélécharge que les derniers jours (+ chevauchement) ;
- les cours sont ajustés des dividendes/splits : si le chevauchement ne colle plus avec le
  cache, tout l'historique du titre est retéléchargé ;
- les titres sans aucun cours sont mémorisés comme « morts » pendant `dead_ttl_hours`, mais
  seulement si l'échec est rare (sinon c'est probablement le réseau, pas le titre) ;
- l'écriture se fait par lots : un scan interrompu garde ce qui a déjà été téléchargé.
"""
from __future__ import annotations

import json
import os
import re
import tempfile
from datetime import datetime
from pathlib import Path
from typing import Callable

import numpy as np
import pandas as pd

_UNSAFE = re.compile(r"[^A-Za-z0-9._-]")

# network(tickers, start, end, progress_cb) -> DataFrame de cours (colonnes = tickers)
Network = Callable[[list, str, str, "Callable[[float], None] | None"], pd.DataFrame]


def _atomic_write(path: Path, text: str) -> None:
    path.parent.mkdir(parents=True, exist_ok=True)
    fd, tmp = tempfile.mkstemp(dir=path.parent, suffix=".tmp")
    try:
        with os.fdopen(fd, "w", encoding="utf-8", newline="") as f:
            f.write(text)
        os.replace(tmp, path)
    except BaseException:
        try:
            os.unlink(tmp)
        except OSError:
            pass
        raise


class DiskCache:
    META = "prices_meta.json"
    PE = "pe.json"
    ROIC = "roic.json"

    def __init__(self, root, ttl_hours: float = 12.0, dead_ttl_hours: float = 24.0,
                 pe_ttl_hours: float = 24.0, overlap_days: int = 7, refresh: bool = False,
                 batch: int = 200, tolerance: float = 1e-3, dead_max_share: float = 0.15,
                 roic_ttl_hours: float = 168.0,
                 now: Callable[[], datetime] | None = None):
        self.root = Path(root)
        self.ttl = ttl_hours
        self.dead_ttl = dead_ttl_hours
        self.pe_ttl = pe_ttl_hours
        self.roic_ttl = roic_ttl_hours      # les comptes changent au trimestre : 7 jours
        self.overlap = pd.Timedelta(days=overlap_days)
        self.refresh = refresh
        self.batch = batch
        self.tolerance = tolerance          # écart relatif toléré sur le chevauchement
        self.dead_max_share = dead_max_share
        self._now = now or datetime.now
        self.stats: dict[str, int] = {}
        self._pe: dict | None = None
        self._roic: dict | None = None

    # ------------------------------------------------------------------ fichiers
    def _series_path(self, ticker: str) -> Path:
        # préfixe : « CON », « NUL »... sont des noms réservés sous Windows
        return self.root / "prices" / f"px_{_UNSAFE.sub('_', ticker)}.csv"

    def _load_meta(self) -> dict:
        try:
            data = json.loads((self.root / self.META).read_text(encoding="utf-8"))
            return data if isinstance(data, dict) else {}
        except (OSError, ValueError):
            return {}

    def _save_meta(self, meta: dict) -> None:
        _atomic_write(self.root / self.META, json.dumps(meta, indent=0, sort_keys=True))

    def _load_series(self, ticker: str) -> pd.Series | None:
        try:
            df = pd.read_csv(self._series_path(ticker), index_col=0, parse_dates=True)
            s = df["close"].dropna()
        except (OSError, ValueError, KeyError, pd.errors.ParserError):
            return None
        if s.empty or not isinstance(s.index, pd.DatetimeIndex):
            return None
        s = s.sort_index().astype(float)
        s.index.name = None  # même index qu'un téléchargement direct
        return s

    def _save_series(self, ticker: str, s: pd.Series) -> None:
        _atomic_write(self._series_path(ticker),
                      s.rename("close").to_csv(index_label="date"))

    # -------------------------------------------------------------------- cours
    def fetch_prices(self, tickers: list, start: str, end: str, network: Network,
                     progress_cb: Callable[[float], None] | None = None) -> pd.DataFrame:
        """Cours de clôture des `tickers` sur [start, end[, depuis le disque quand c'est possible."""
        start_ts, end_ts = pd.Timestamp(start), pd.Timestamp(end)
        now = self._now()
        stamp = now.isoformat(timespec="seconds")
        meta = self._load_meta()
        self.stats = dict.fromkeys(("fresh", "updated", "full", "dead", "redo", "missing"), 0)

        series: dict[str, pd.Series] = {}
        full: list[str] = []
        incr: list[str] = []
        for t in tickers:
            m = meta.get(t)
            if self.refresh or not isinstance(m, dict):
                full.append(t)
                continue
            try:
                age = (now - datetime.fromisoformat(m["fetched_at"])).total_seconds() / 3600
                if m.get("status") == "empty":
                    if age < self.dead_ttl:
                        self.stats["dead"] += 1
                    else:
                        full.append(t)
                    continue
                if pd.Timestamp(m["covered_from"]) > start_ts:
                    full.append(t)          # fenêtre demandée plus longue que celle en cache
                    continue
            except (KeyError, ValueError, TypeError):
                full.append(t)
                continue
            s = self._load_series(t)
            if s is None:
                full.append(t)
                continue
            series[t] = s
            if age < self.ttl:
                self.stats["fresh"] += 1
            else:
                incr.append(t)

        total = max(1, len(tickers))
        state = {"done": len(tickers) - len(full) - len(incr)}
        if progress_cb and state["done"]:
            progress_cb(state["done"] / total)

        def call(part: list, g_start: str) -> pd.DataFrame:
            base = state["done"]
            cb = (lambda f: progress_cb((base + f * len(part)) / total)) if progress_cb else None
            df = network(part, g_start, end, cb)
            state["done"] += len(part)
            return df if df is not None else pd.DataFrame()

        def store_full(part: list, df: pd.DataFrame) -> list[str]:
            """Enregistre les titres reçus ; retourne ceux qui n'ont rien renvoyé."""
            missing = []
            for t in part:
                s = df[t].dropna() if t in df.columns else pd.Series(dtype=float)
                if s.empty:
                    missing.append(t)
                    continue
                self._save_series(t, s)
                meta[t] = {"status": "ok", "fetched_at": stamp,
                           "covered_from": start_ts.strftime("%Y-%m-%d")}
                series[t] = s
            return missing

        # 1) titres inconnus / à recharger en entier, par lots (progression sauvegardée)
        no_data: list[str] = []
        for i in range(0, len(full), self.batch):
            part = full[i:i + self.batch]
            no_data += store_full(part, call(part, start))
            self.stats["full"] += len(part)
            self._save_meta(meta)

        # 2) titres en cache mais périmés : seulement les derniers jours
        redo: list[str] = []
        for i in range(0, len(incr), self.batch):
            part = incr[i:i + self.batch]
            g_start = (min(series[t].index[-1] for t in part) - self.overlap).strftime("%Y-%m-%d")
            df = call(part, g_start)
            for t in part:
                new = df[t].dropna() if t in df.columns else pd.Series(dtype=float)
                if new.empty:
                    continue  # échec ou rien de neuf : on garde le cache, retenté au prochain run
                old = series[t]
                common = old.index.intersection(new.index)
                if common.empty or ((new[common] / old[common].replace(0, np.nan) - 1)
                                    .abs().max() > self.tolerance):
                    redo.append(t)   # ajustement (dividende, split) ou trou : tout retélécharger
                    continue
                merged = pd.concat([old[old.index < new.index.min()], new])
                self._save_series(t, merged)
                series[t] = merged
                meta[t] = {**meta[t], "fetched_at": stamp}
                self.stats["updated"] += 1
            self._save_meta(meta)

        # 3) historiques à refaire : on ne remplace le cache que si le téléchargement réussit
        for i in range(0, len(redo), self.batch):
            part = redo[i:i + self.batch]
            store_full(part, call(part, start))
            self.stats["redo"] += len(part)
            self._save_meta(meta)

        # 4) titres sans cours : mémorisés « morts » seulement si l'échec est rare
        self.stats["missing"] = len(no_data)
        never_ok = [t for t in no_data if meta.get(t, {}).get("status") != "ok"]
        if full and len(no_data) <= self.dead_max_share * len(full):
            for t in never_ok:
                meta[t] = {"status": "empty", "fetched_at": stamp}
            if never_ok:
                self._save_meta(meta)

        cols = [t for t in tickers if t in series]
        if not cols:
            return pd.DataFrame()
        df = pd.concat({t: series[t] for t in cols}, axis=1).sort_index()
        return df[(df.index >= start_ts) & (df.index < end_ts)]

    # ---------------------------------------------------------------------- PER
    def _pe_data(self) -> dict:
        if self._pe is None:
            try:
                data = json.loads((self.root / self.PE).read_text(encoding="utf-8"))
                self._pe = data if isinstance(data, dict) else {}
            except (OSError, ValueError):
                self._pe = {}
        return self._pe

    def get_pe(self, ticker: str) -> tuple[float, float] | None:
        """(PER 12 mois, PER prévu) en cache et encore valable, sinon None."""
        if self.refresh:
            return None
        e = self._pe_data().get(ticker)
        try:
            age = (self._now() - datetime.fromisoformat(e["fetched_at"])).total_seconds() / 3600
            if age >= self.pe_ttl:
                return None
            return (np.nan if e["trailing"] is None else float(e["trailing"]),
                    np.nan if e["forward"] is None else float(e["forward"]))
        except (KeyError, ValueError, TypeError):
            return None

    def save_pe(self, items: dict) -> None:
        """Enregistre {ticker: (PER, PER prévu)} ; NaN devient null."""
        if not items:
            return
        data = self._pe_data()
        stamp = self._now().isoformat(timespec="seconds")
        for t, (tr, fw) in items.items():
            data[t] = {"trailing": None if pd.isna(tr) else float(tr),
                       "forward": None if pd.isna(fw) else float(fw),
                       "fetched_at": stamp}
        _atomic_write(self.root / self.PE, json.dumps(data, indent=0, sort_keys=True))

    # --------------------------------------------------------------------- ROIC
    def _roic_data(self) -> dict:
        if self._roic is None:
            try:
                data = json.loads((self.root / self.ROIC).read_text(encoding="utf-8"))
                self._roic = data if isinstance(data, dict) else {}
            except (OSError, ValueError):
                self._roic = {}
        return self._roic

    def get_roic(self, ticker: str) -> tuple[float, float] | None:
        """(ROIC dernier exercice, ROIC moyen) en cache et encore valable, sinon None."""
        if self.refresh:
            return None
        e = self._roic_data().get(ticker)
        try:
            age = (self._now() - datetime.fromisoformat(e["fetched_at"])).total_seconds() / 3600
            if age >= self.roic_ttl:
                return None
            return (np.nan if e["last"] is None else float(e["last"]),
                    np.nan if e["avg"] is None else float(e["avg"]))
        except (KeyError, ValueError, TypeError):
            return None

    def save_roic(self, items: dict) -> None:
        """Enregistre {ticker: (ROIC, ROIC moyen)} ; NaN devient null."""
        if not items:
            return
        data = self._roic_data()
        stamp = self._now().isoformat(timespec="seconds")
        for t, (last, avg) in items.items():
            data[t] = {"last": None if pd.isna(last) else float(last),
                       "avg": None if pd.isna(avg) else float(avg),
                       "fetched_at": stamp}
        _atomic_write(self.root / self.ROIC, json.dumps(data, indent=0, sort_keys=True))

===== cli.py =====
"""
Version ligne de commande du scanner de drawdown XTB (yfinance).

Installation :  pip install yfinance pandas numpy
Exemples :
    python drawdown_cli.py 20 3m                 # drawdown >= 20 % sur 3 mois
    python drawdown_cli.py 30 1y --mode max --pea --above-ma200
    python drawdown_cli.py 25 6m --pe-max 15     # + filtre PER <= 15 (sur la short-list)
    python drawdown_cli.py 25 6m --roic-min 10   # + filtre ROIC moyen >= 10 % (sur la short-list)
    python drawdown_cli.py                       # mode interactif (questions)
    python drawdown_cli.py --resolve             # vérifie/corrige la conversion XTB -> Yahoo
    python drawdown_cli.py 30 6m --cache-ttl 72  # cours réutilisés jusqu'à 72 h sans requête
    python drawdown_cli.py 30 6m --refresh       # ignore le cache et retélécharge tout
    python drawdown_cli.py --resolve --markets SE DK FI --suggest

Par défaut la sortie est la seule liste des titres retenus (une ligne « ticker  nom », le plus
touché d'abord) ; -v / --verbose rétablit la progression, le bilan du cache et le tableau détaillé.
Les CFD sont toujours exclus.
Durée : 90d (jours), 12w (semaines), 3m (mois), 1y (ans). "j" et "a" acceptés.
"""
from __future__ import annotations

import argparse
import logging
import re
import sys
from datetime import datetime
from pathlib import Path

import pandas as pd

import scanner as ds

UNITS = {"d": "days", "j": "days", "w": "weeks", "m": "months", "y": "years", "a": "years"}


def parse_duration(text: str) -> pd.DateOffset | pd.Timedelta:
    m = re.fullmatch(r"\s*(\d+)\s*([a-zA-Z]+)\s*", text)
    if not m or m.group(2).lower() not in UNITS:
        raise argparse.ArgumentTypeError("durée invalide, ex. : 90d, 12w, 3m, 1y")
    n, unit = int(m.group(1)), UNITS[m.group(2).lower()]
    if unit == "days":
        return pd.Timedelta(days=n)
    if unit == "weeks":
        return pd.Timedelta(weeks=n)
    return pd.DateOffset(**{unit: n})


# --- Export du projet pour un LLM -------------------------------------------
DUMP_NAME = "forllm.txt"
DUMP_EXTS = {".py", ".md", ".toml", ".cfg", ".ini", ".json", ".yaml", ".yml", ".sh", ".txt"}
DUMP_SKIP_DIRS = {"__pycache__", "venv", ".venv", "env", "node_modules", "build", "dist"}
DUMP_MAX_BYTES = 500_000  # ignore les gros fichiers (données, exports)
DUMP_SKIP_NAMES = {ds.MAP_FILE}  # table de conversion : donnée, pas du code


def dump_for_llm(root: Path | None = None, name: str = DUMP_NAME) -> Path:
    """Concatène tous les fichiers texte du projet dans <root>/forllm.txt."""
    root = (root or Path(__file__).resolve().parent).resolve()
    target = root / name
    files = []
    for f in sorted(root.rglob("*")):
        rel = f.relative_to(root)
        if not f.is_file() or f == target:
            continue
        if any(part.startswith(".") or part in DUMP_SKIP_DIRS for part in rel.parts[:-1]):
            continue
        if f.name.startswith(".") or f.name in DUMP_SKIP_NAMES or f.suffix.lower() not in DUMP_EXTS:
            continue
        if f.stat().st_size > DUMP_MAX_BYTES:
            continue
        files.append(f)

    parts = [f"# Projet : {root.name}",
             f"# Généré le {datetime.now():%Y-%m-%d %H:%M}",
             "# Fichiers : " + ", ".join(f.relative_to(root).as_posix() for f in files),
             ""]
    for f in files:
        try:
            text = f.read_text(encoding="utf-8")
        except (UnicodeDecodeError, OSError):
            continue  # binaire ou illisible
        parts.append(f"===== {f.relative_to(root).as_posix()} =====")
        parts.append(text.rstrip("\n"))
        parts.append("")
    target.write_text("\n".join(parts), encoding="utf-8")
    return target

def run_resolve(uni: pd.DataFrame, map_path: str, suggest: bool) -> int:
    print(f"{len(uni)} tickers à vérifier (cours des 21 derniers jours)…", file=sys.stderr)
    try:
        out = ds.resolve_tickers(uni, suggest=suggest)
    except RuntimeError as exc:
        print(f"Erreur : {exc}", file=sys.stderr)
        return 1
    ds.save_ticker_map(map_path, out["resolved"], out["unresolved"], out["checked"])
    resolved, unresolved = out["resolved"], out["unresolved"]
    print(f"{len(uni)} titres vérifiés : {out['ok']} OK, {len(resolved)} corrigés "
          f"automatiquement, {len(unresolved)} sans correspondance")
    if resolved:
        print("\nCorrigés :")
        for xtb, y in sorted(resolved.items()):
            print(f"  {xtb:<14} -> {y}")
    if unresolved:
        print("\nSans correspondance (corrige à la main dans \"overrides\" du JSON) :")
        for xtb, info in sorted(unresolved.items()):
            print(f"  {xtb:<14} {info['desc'][:40]}")
            for sug in info["suggestions"]:
                print(f"{'':18}? {sug}")
    print(f"\nTable enregistrée : {map_path}")
    return 0


def main() -> int:
    args = build_parser().parse_args()
    if not args.no_cache and not args.resolve:
        root = args.cache_dir or Path(args.csv).resolve().parent / ds.CACHE_DIR
        ds.configure_cache(root, ttl_hours=args.cache_ttl, refresh=args.refresh)
    yf_log = logging.getLogger("yfinance")
    previous_level = yf_log.level
    if not args.verbose:
        yf_log.setLevel(logging.CRITICAL)  # plus de « possibly delisted » ni d'erreurs curl
    try:
        return _main(args)
    finally:
        yf_log.setLevel(previous_level)
        ds.configure_cache(None)  # pas d'état global qui traîne entre deux appels


def _main(args) -> int:
    if args.drawdown is None and not args.resolve:
        args.drawdown = float(input("Drawdown minimum (%) : ").replace(",", "."))
    if args.duree is None and not args.resolve:
        args.duree = input("Durée (ex. 3m, 90d, 1y) : ")
    map_path = args.map or str(Path(args.csv).with_name(ds.MAP_FILE))

    try:
        uni = ds.load_universe(args.csv, ds.load_ticker_map(map_path))
    except FileNotFoundError:
        sys.exit(f"CSV introuvable : {args.csv} (option --csv)")
    uni = uni[uni["section"].isin(args.types) & ~uni["close_only"]]
    if args.markets:
        uni = uni[uni["market"].isin([m.upper() for m in args.markets])]
    if not args.with_leveraged:
        uni = uni[~uni["is_leveraged"]]
    if args.pea:
        uni = uni[uni["pea_probable"]]
    if args.limit:
        uni = uni.head(args.limit)
    uni = uni.reset_index(drop=True)
    if uni.empty:
        sys.exit("Aucun titre après filtres.")
    if args.resolve:
        return run_resolve(uni, map_path, args.suggest)
    offset = parse_duration(args.duree)
    dd_threshold = abs(args.drawdown)

    end = pd.Timestamp.today().normalize()
    window_start = end - offset
    col = "DD actuel %" if args.mode == "actuel" else "DD max %"
    verbose = args.verbose
    if verbose:
        print(f"{len(uni)} titres, fenêtre depuis {window_start.date()}, "
              f"drawdown {args.mode} <= -{dd_threshold:g} %. Téléchargement…", file=sys.stderr)

    def progress(p):
        print(f"\r  {p:4.0%}", end="", file=sys.stderr, flush=True)

    def pe_progress(p):
        print(f"\r  PER {p:4.0%}", end="", file=sys.stderr, flush=True)

    def roic_progress(p):
        print(f"\r  ROIC {p:4.0%}", end="", file=sys.stderr, flush=True)

    res, _, missing, analysed = ds.run_scan(
        uni, window_start, end, dd_threshold, col, args.above_ma200,
        progress if verbose else None,
        pe_max=args.pe_max, pe_min=args.pe_min,
        pe_col="PER" if args.pe_type == "trailing" else "PER prévu",
        keep_missing_pe=args.keep_no_pe, show_pe=args.show_pe,
        pe_progress_cb=pe_progress if verbose else None,
        roic_min=args.roic_min,
        roic_col="ROIC moy %" if args.roic_type == "avg" else "ROIC %",
        keep_missing_roic=args.keep_no_roic, show_roic=args.show_roic,
        roic_progress_cb=roic_progress if verbose else None)
    if verbose:
        print(file=sys.stderr)
        if ds.CACHE and ds.CACHE.stats:
            st = ds.CACHE.stats
            print(f"Cache : {st['fresh']} à jour, {st['updated']} complétés, "
                  f"{st['full'] + st['redo']} téléchargés en entier, {st['dead']} ignorés "
                  f"(sans cours récemment), {st['missing']} sans réponse.", file=sys.stderr)
        print(f"{len(res)} titres retenus sur {analysed} analysés "
              f"({len(missing)} sans données)\n")
    total = analysed + len(missing)
    if missing:
        if total >= 40 and len(missing) > 0.25 * total:
            # seul message conservé hors mode verbose : la liste serait trompeuse
            print(f"ATTENTION : {len(missing)} titres sur {total} sans données, probablement un "
                  "problème réseau : liste incomplète, relance (-v pour le détail).",
                  file=sys.stderr)
        elif verbose:
            print("Astuce : `--resolve` peut corriger la conversion XTB -> Yahoo "
                  "des titres sans données.", file=sys.stderr)
    if res.empty:
        return 0

    cols = ["ticker", "desc", "currency", "DD actuel %", "DD max %", "Perf période %",
            "vs MM200 %"]
    cols += [c for c in ("PER", "PER prévu", "ROIC %", "ROIC moy %") if c in res.columns]
    cols += ["Dernier cours", "Date plus haut"]
    table = res[cols].rename(columns={"ticker": "Ticker", "desc": "Nom", "currency": "Devise"})
    num = table.select_dtypes("number").columns
    table[num] = table[num].round(2)
    table["Nom"] = table["Nom"].str.slice(0, 32)
    if verbose:
        with pd.option_context("display.width", 200, "display.max_rows", None):
            print(table.to_string(index=False))
    else:
        width = res["ticker"].str.len().max()
        print("\n".join(f"{t:<{width}}  {d}" for t, d in zip(res["ticker"], res["desc"])))
    if args.out:
        table.to_csv(args.out, index=False)
        if verbose:
            print(f"\nExporté : {args.out}")
    return 0

def configure_cache(root, **options) -> DiskCache | None:
    """Active le cache disque dans `root` (None pour le désactiver)."""
    global CACHE
    CACHE = DiskCache(root, **options) if root else None
    return CACHE


# Téléchargement : peu de requêtes en parallèle (trop de threads => timeouts DNS / limites Yahoo
# et des titres existants sont alors pris à tort pour « delisted »), puis seconde chance.
DOWNLOAD_THREADS = 4
RETRY_ROUNDS = 2   # passes de rattrapage sur les titres sans cours
RETRY_CHUNK = 10   # petits lots pendant le rattrapage

# Suffixe de marché XTB -> suffixe Yahoo Finance
YAHOO_SUFFIX = {
    "US": "", "FR": ".PA", "DE": ".DE", "ES": ".MC", "IT": ".MI", "NL": ".AS",
    "BE": ".BR", "PT": ".LS", "FI": ".HE", "SE": ".ST", "DK": ".CO",
    "NO": ".OL", "CH": ".SW", "UK": ".L", "CZ": ".PR",
}
# Marchés de l'EEE (approximation "PEA probable", voir avertissement dans l'UI)
EEA_MARKETS = {"FR", "DE", "ES", "IT", "NL", "BE", "PT", "FI", "SE", "DK", "NO"}

# ETF à effet de levier / inverses, repérés par leur nom (heuristique).
# "Ultra Short-Term Bond" ou "Short Term Treasury" ne sont PAS à levier : ils sont épargnés.
LEVERAGED_RE = re.compile(
    r"\b[1-5]x\b|\bbull\b|\bbear\b|\binverse\b|\bleverag|\blevered\b"
    r"|\bultra(?!\s*short[- ]?term)|proshares\s+short\b",
    re.IGNORECASE,
)

# Corrections manuelles quand la règle générale ne suffit pas (actions de classe B, etc.)
# Complète cette liste avec les tickers listés dans "Sans données" après un scan.
OVERRIDES = {
    "BRKB.US": "BRK-B",
    "ERICB.SE": "ERIC-B.ST",
    "NOVOB.DK": "NOVO-B.CO",
    "VOLVB.SE": "VOLV-B.ST",
}

def to_yahoo(xtb_ticker: str) -> str | None:
    """Convertit un ticker XTB (ex. 'SAN.FR') en ticker Yahoo (ex. 'SAN.PA')."""
    if xtb_ticker in OVERRIDES:
        return OVERRIDES[xtb_ticker]
    if "." not in xtb_ticker:
        return None
    base, market = xtb_ticker.rsplit(".", 1)
    if market not in YAHOO_SUFFIX:
        return None
    return base.replace(".", "-") + YAHOO_SUFFIX[market]


# --------------------------------------------------------------------------- #
# Conversion XTB -> Yahoo : candidats, validation, table de correspondance
# --------------------------------------------------------------------------- #
MAP_FILE = "ticker_map.json"
# Classes d'actions écrites avec un tiret chez Yahoo (NOVO-B.CO, ERIC-B.ST, BRK-B)
DASH_CLASS_LETTERS = {"US": "ABC", "DK": "ABCDKRV", "SE": "ABCDKRV",
                      "NO": "ABCDKRV", "FI": "ABCDKRV"}
ALT_SUFFIX = {"DE": [".F"]}  # place de repli quand Xetra n'a rien


def candidates(xtb_ticker: str) -> list[str]:
    """Tickers Yahoo possibles pour un ticker XTB, du plus au moins probable.
    Le premier est celui de to_yahoo(). Les suivants ne servent que si lui-même
    ne renvoie aucun cours (voir resolve_tickers)."""
    first = to_yahoo(xtb_ticker)
    if first is None:
        return []
    if xtb_ticker in OVERRIDES:
        return [first]
    base, market = xtb_ticker.rsplit(".", 1)
    base = base.replace(".", "-")
    suffix = YAHOO_SUFFIX[market]
    out = [first]
    letters = DASH_CLASS_LETTERS.get(market, "")
    if letters and len(base) >= 3 and "-" not in base and base[-1] in letters:
        out.append(f"{base[:-1]}-{base[-1]}{suffix}")  # NOVOB -> NOVO-B
    out += [base + alt for alt in ALT_SUFFIX.get(market, [])]
    return list(dict.fromkeys(out))


def load_ticker_map(path) -> dict[str, str]:
    """Corrections XTB -> Yahoo enregistrées dans le fichier JSON (modifiables à la main)."""
    p = Path(path)
    if not p.exists():
        return {}
    try:
        data = json.loads(p.read_text(encoding="utf-8"))
    except (OSError, ValueError):
        return {}
    overrides = data.get("overrides", {}) if isinstance(data, dict) else {}
    return {str(k): str(v) for k, v in overrides.items() if v}


def save_ticker_map(path, resolved: dict, unresolved: dict, checked) -> None:
    """Fusionne les résultats d'une résolution dans le JSON. Les corrections déjà
    présentes (donc éventuellement faites à la main) ne sont jamais écrasées ; la liste
    « unresolved » garde les titres non revérifiés cette fois-ci."""
    p = Path(path)
    overrides = load_ticker_map(p)
    overrides.update({k: v for k, v in resolved.items() if k not in overrides})
    prev = {}
    if p.exists():
        try:
            prev = json.loads(p.read_text(encoding="utf-8")).get("unresolved", {})
        except (OSError, ValueError, AttributeError):
            prev = {}
    keep = {k: v for k, v in prev.items() if k not in set(checked) and k not in overrides}
    keep.update(unresolved)
    data = {"updated": date.today().isoformat(),
            "overrides": dict(sorted(overrides.items())),
            "unresolved": dict(sorted(keep.items()))}
    p.write_text(json.dumps(data, ensure_ascii=False, indent=2), encoding="utf-8")


def _alive(tickers: list[str], days: int, chunk: int,
           retry_rounds: int = RETRY_ROUNDS, guard: bool = True) -> set[str]:
    """Tickers Yahoo ayant au moins un cours sur les `days` derniers jours.
    Les titres sans cours sont retentés en petits lots avant d'être déclarés morts.
    Lève RuntimeError si Yahoo ne répond pas (ou, avec guard, si plus d'un quart des titres
    n'ont pas de cours : problème réseau probable), pour ne pas enregistrer une table fausse."""
    end = pd.Timestamp.today().normalize()
    start_s = (end - pd.Timedelta(days=days)).strftime("%Y-%m-%d")
    end_s = (end + pd.Timedelta(days=1)).strftime("%Y-%m-%d")
    alive: set[str] = set()
    for i in range(0, len(tickers), chunk):
        part = tuple(tickers[i:i + chunk])
        got = None
        for _ in range(3):
            try:
                got = {str(c) for c in _download_close(part, start_s, end_s).columns}
            except Exception:
                got = None
            if got:
                break
            time.sleep(2)
        if got is None:
            raise RuntimeError("Yahoo injoignable : abandon pour ne pas enregistrer une table fausse.")
        if not got and len(part) >= 10:
            raise RuntimeError("Yahoo n'a renvoyé aucun cours pour tout un lot "
                               "(limite de débit ?). Réessaie plus tard.")
        alive |= got

    dead = [t for t in tickers if t not in alive]
    for rnd in range(retry_rounds):
        if not dead:
            break
        time.sleep(2 * (rnd + 1))
        for i in range(0, len(dead), RETRY_CHUNK):
            try:
                alive |= {str(c) for c in
                          _download_close(tuple(dead[i:i + RETRY_CHUNK]), start_s, end_s).columns}
            except Exception:
                pass
        dead = [t for t in tickers if t not in alive]
    if guard and len(tickers) >= 40 and len(dead) > 0.25 * len(tickers):
        raise RuntimeError(f"{len(dead)}/{len(tickers)} tickers sans cours : probablement un "
                           "problème réseau (DNS ?), pas des titres disparus. Abandon pour ne "
                           "pas enregistrer une table fausse.")
    return alive


def _suggest(desc: str, limit: int = 3) -> list[str]:
    """Pistes de tickers Yahoo à partir du nom (à vérifier à la main, jamais appliquées seules)."""
    search = getattr(yf, "Search", None)
    if search is None or not desc:
        return []
    try:
        quotes = search(desc, max_results=limit).quotes
    except Exception:
        return []
    return [f"{q['symbol']} ({q.get('exchDisp') or q.get('exchange') or '?'}, "
            f"{q.get('shortname') or q.get('longname') or '?'})"
            for q in quotes if q.get("symbol")]


def resolve_tickers(universe: pd.DataFrame, days: int = 21, chunk: int = 100,
                    suggest: bool = False, retry_rounds: int = RETRY_ROUNDS) -> dict:
    """Vérifie chaque ticker Yahoo de l'univers (colonnes ticker, desc, yahoo) puis, pour
    ceux sans cours récent, essaie les candidats alternatifs de candidates().
    Retourne {"checked", "ok", "resolved": {xtb: yahoo}, "unresolved": {xtb: {...}}}."""
    alive = _alive(universe["yahoo"].tolist(), days, chunk, retry_rounds)
    dead = universe[~universe["yahoo"].isin(alive)]
    alternatives = {xtb: [c for c in candidates(xtb) if c != y]
                    for xtb, y in zip(dead["ticker"], dead["yahoo"])}
    pool = sorted({c for alts in alternatives.values() for c in alts})
    # alternatives : la plupart sont mortes par construction, donc ni garde-fou ni insistance
    alive_alt = _alive(pool, days, chunk, min(retry_rounds, 1), guard=False) if pool else set()
    taken = set(universe["yahoo"])  # un ticker Yahoo ne doit servir qu'à un seul titre XTB
    resolved, unresolved = {}, {}
    for xtb, y, desc in zip(dead["ticker"], dead["yahoo"], dead["desc"]):
        hit = next((c for c in alternatives[xtb] if c in alive_alt and c not in taken), None)
        if hit:
            resolved[xtb] = hit
            taken.add(hit)
        else:
            unresolved[xtb] = {"desc": desc, "tried": [y, *alternatives[xtb]],
                               "suggestions": _suggest(desc) if suggest else []}
    return {"checked": set(universe["ticker"]), "ok": len(alive & set(universe["yahoo"])),
            "resolved": resolved, "unresolved": unresolved}


def load_universe(src, ticker_map: dict | None = None) -> pd.DataFrame:
    df = pd.read_csv(src)
    df["ticker"] = df["ticker"].astype(str).str.strip()
    df["desc"] = df["desc"].fillna("").astype(str)
    df["close_only"] = df["ticker"].str.endswith("*")  # titres en "CLOSE ONLY"
    df["ticker"] = df["ticker"].str.rstrip("*")
    df["market"] = df["ticker"].str.rsplit(".", n=1).str[-1]
    df = df[~df["desc"].str.contains("CFD", case=False)].copy()  # CFD : toujours exclus
    df["is_leveraged"] = (df["section"] == "ETF") & df["desc"].str.contains(LEVERAGED_RE)
    tmap = ticker_map or {}
    df["yahoo"] = df["ticker"].map(lambda t: tmap.get(t) or to_yahoo(t))
    df["pea_probable"] = df["market"].isin(EEA_MARKETS) & (
        (df["section"] != "ETF") | df["desc"].str.upper().str.contains("PEA")
    )
    df = df.dropna(subset=["yahoo"]).sort_values("close_only")
    df = df.drop_duplicates("ticker").drop_duplicates("yahoo")
    return df.reset_index(drop=True)

def _cache(func):
    return st.cache_data(show_spinner=False, ttl=3600)(func) if st else func


def _download_close(tickers: tuple, start: str, end: str) -> pd.DataFrame:
    data = yf.download(
        list(tickers), start=start, end=end,
        auto_adjust=True, progress=False, threads=DOWNLOAD_THREADS,
    )
    if data is None or data.empty:
        return pd.DataFrame()
    close = data["Close"]
    if isinstance(close, pd.Series):  # un seul ticker, anciennes versions
        close = close.to_frame(tickers[0])
    return close.dropna(axis=1, how="all")


fetch_chunk = _cache(_download_close)


def fetch_prices(tickers, start, end, progress_cb=None, chunk=100,
                 retry_rounds=RETRY_ROUNDS) -> pd.DataFrame:
    """Cours de clôture ajustés. Passe par le cache disque s'il est activé."""
    if CACHE is None:
        return _fetch_network(tickers, start, end, progress_cb, chunk, retry_rounds)

    def network(part, g_start, g_end, cb):
        return _fetch_network(part, g_start, g_end, cb, chunk, retry_rounds)
    return CACHE.fetch_prices(list(tickers), start, end, network, progress_cb)


def _fetch_network(tickers, start, end, progress_cb=None, chunk=100,
                   retry_rounds=RETRY_ROUNDS) -> pd.DataFrame:
    frames = []
    for i in range(0, len(tickers), chunk):
        part = tuple(tickers[i:i + chunk])
        for _ in range(2):  # un essai de plus en cas d'erreur réseau / limite Yahoo
            try:
                frames.append(fetch_chunk(part, start, end))
                break
            except Exception:
                time.sleep(2)
        if progress_cb:
            progress_cb(min(i + chunk, len(tickers)) / len(tickers))
    prices = pd.concat(frames, axis=1) if frames else pd.DataFrame()
    prices = prices.loc[:, ~prices.columns.duplicated()]

    # Seconde chance : yfinance ne lève pas d'erreur quand une requête échoue (timeout DNS...),
    # le titre manque simplement. On retente ceux-là en petits lots ; un vrai titre mort
    # échoue vite et reste manquant.
    missing = [t for t in tickers if t not in prices.columns]
    for rnd in range(retry_rounds):
        if not missing:
            break
        time.sleep(2 * (rnd + 1))
        extra = []
        for i in range(0, len(missing), RETRY_CHUNK):
            try:
                extra.append(fetch_chunk(tuple(missing[i:i + RETRY_CHUNK]), start, end))
            except Exception:
                pass
        extra = [f for f in extra if not f.empty]
        if extra:
            prices = pd.concat([prices, *extra], axis=1)
            prices = prices.loc[:, ~prices.columns.duplicated()]
        missing = [t for t in tickers if t not in prices.columns]

    if prices.empty:
        return pd.DataFrame()
    idx = pd.to_datetime(prices.index)
    prices.index = idx.tz_localize(None) if idx.tz is not None else idx
    return prices.sort_index()


# --------------------------------------------------------------------------- #
# PER (Price/Earnings)
# --------------------------------------------------------------------------- #
_PE_CACHE: dict[str, tuple[float, float]] = {}
_PE_NEW: dict[str, tuple[float, float]] = {}  # reçus du réseau, à écrire sur disque


def _to_float(x) -> float:
    try:
        v = float(x)
        return v if np.isfinite(v) else np.nan
    except (TypeError, ValueError):
        return np.nan


def get_pe(yahoo_ticker: str) -> tuple[float, float]:
    """(PER sur 12 derniers mois, PER prévu). NaN si Yahoo n'a pas la donnée
    (bénéfices négatifs, ETF, titre mal couvert...)."""
    if yahoo_ticker not in _PE_CACHE:
        hit = CACHE.get_pe(yahoo_ticker) if CACHE else None
        if hit is not None:
            _PE_CACHE[yahoo_ticker] = hit
        else:
            try:
                info = yf.Ticker(yahoo_ticker).info
                if not info:
                    raise ValueError("réponse vide")
                _PE_CACHE[yahoo_ticker] = (_to_float(info.get("trailingPE")),
                                           _to_float(info.get("forwardPE")))
                _PE_NEW[yahoo_ticker] = _PE_CACHE[yahoo_ticker]
            except Exception:  # échec : retenu pour la session seulement, jamais sur disque
                _PE_CACHE[yahoo_ticker] = (np.nan, np.nan)
    return _PE_CACHE[yahoo_ticker]


def add_pe(res: pd.DataFrame, progress_cb=None, workers: int = 8) -> pd.DataFrame:
    """Ajoute les colonnes PER / PER prévu (une requête par titre, en parallèle).
    Appelé uniquement sur la short-list après filtre drawdown, pour rester rapide.
    Le PER n'a de sens que pour les actions : les ETF ne sont pas interrogés (NaN)."""
    tickers = res.loc[res["section"] == "Stock", "yahoo"].tolist()
    out = {}
    with ThreadPoolExecutor(max_workers=workers) as ex:
        futs = {ex.submit(get_pe, y): y for y in tickers}
        for i, f in enumerate(as_completed(futs), 1):
            out[futs[f]] = f.result()
            if progress_cb:
                progress_cb(i / len(tickers))
    if CACHE and _PE_NEW:
        CACHE.save_pe(dict(_PE_NEW))
        _PE_NEW.clear()
    res = res.copy()
    res["PER"] = res["yahoo"].map(lambda y: out.get(y, (np.nan, np.nan))[0])
    res["PER prévu"] = res["yahoo"].map(lambda y: out.get(y, (np.nan, np.nan))[1])
    return res


def filter_pe(res: pd.DataFrame, pe_min: float, pe_max: float,
              pe_col: str, keep_missing: bool) -> pd.DataFrame:
    """Filtre sur le PER. S'applique aux actions uniquement : les ETF passent toujours."""
    pe = res[pe_col]
    ok = pe.between(pe_min, pe_max)  # NaN -> False
    if keep_missing:
        ok = ok | pe.isna()
    ok = ok | (res["section"] != "Stock")
    return res[ok]


# --------------------------------------------------------------------------- #
# ROIC (rentabilité du capital investi)
# --------------------------------------------------------------------------- #
_ROIC_CACHE: dict[str, tuple[float, float]] = {}
_ROIC_NEW: dict[str, tuple[float, float]] = {}  # reçus du réseau, à écrire sur disque
ROIC_DEFAULT_TAX = 0.25   # taux d'impôt de repli quand Yahoo n'en fournit pas
ROIC_MAX_YEARS = 5


def _first_row(df: pd.DataFrame, names: tuple[str, ...]) -> pd.Series | None:
    """Première ligne trouvée parmi `names` (les libellés varient selon les versions de yfinance)."""
    if df is None or df.empty:
        return None
    for n in names:
        if n in df.index:
            return pd.to_numeric(df.loc[n], errors="coerce")
    return None


def compute_roic(income: pd.DataFrame, balance: pd.DataFrame) -> tuple[float, float]:
    """(ROIC du dernier exercice, ROIC moyen) en %, ou (NaN, NaN) si les comptes sont inexploitables.

    ROIC = résultat opérationnel x (1 - taux d'impôt) / capital investi, par exercice annuel.
    Capital investi : ligne « Invested Capital » de Yahoo, sinon dette + capitaux propres - trésorerie.
    Les exercices où le capital investi est <= 0 sont ignorés (ratio non significatif).
    La moyenne porte sur les ROIC_MAX_YEARS derniers exercices disponibles."""
    ebit = _first_row(income, ("Operating Income", "EBIT"))
    if ebit is None:
        return np.nan, np.nan
    tax = _first_row(income, ("Tax Rate For Calcs",))
    invested = _first_row(balance, ("Invested Capital",))
    if invested is None:
        debt = _first_row(balance, ("Total Debt",))
        equity = _first_row(balance, ("Stockholders Equity", "Total Equity Gross Minority Interest"))
        cash = _first_row(balance, ("Cash And Cash Equivalents",
                                    "Cash Cash Equivalents And Short Term Investments"))
        if debt is None or equity is None:
            return np.nan, np.nan
        invested = debt + equity
        if cash is not None:
            invested = invested.sub(cash.reindex(invested.index).fillna(0.0))
    ebit = ebit.copy()
    ebit.index = pd.to_datetime(ebit.index, errors="coerce")
    invested = invested.copy()
    invested.index = pd.to_datetime(invested.index, errors="coerce")
    ebit = ebit[ebit.index.notna()]
    invested = invested[invested.index.notna()]
    if tax is not None:
        tax = tax.copy()
        tax.index = pd.to_datetime(tax.index, errors="coerce")
        tax = tax.reindex(ebit.index)
        tax = tax.where((tax >= 0) & (tax <= 0.5)).fillna(ROIC_DEFAULT_TAX)
    else:
        tax = pd.Series(ROIC_DEFAULT_TAX, index=ebit.index)
    inv = invested.reindex(ebit.index)
    roic = (ebit * (1 - tax) / inv.where(inv > 0) * 100).replace([np.inf, -np.inf], np.nan).dropna()
    if roic.empty:
        return np.nan, np.nan
    roic = roic.sort_index().tail(ROIC_MAX_YEARS)
    return float(roic.iloc[-1]), float(roic.mean())


def get_roic(yahoo_ticker: str) -> tuple[float, float]:
    """(ROIC dernier exercice %, ROIC moyen %). NaN si Yahoo n'a pas les comptes
    (ETF, titre mal couvert, capital investi négatif...)."""
    if yahoo_ticker not in _ROIC_CACHE:
        hit = CACHE.get_roic(yahoo_ticker) if CACHE else None
        if hit is not None:
            _ROIC_CACHE[yahoo_ticker] = hit
        else:
            try:
                tk = yf.Ticker(yahoo_ticker)
                income, balance = tk.income_stmt, tk.balance_sheet
                if income is None or income.empty or balance is None or balance.empty:
                    raise ValueError("comptes vides")
                _ROIC_CACHE[yahoo_ticker] = compute_roic(income, balance)
                _ROIC_NEW[yahoo_ticker] = _ROIC_CACHE[yahoo_ticker]
            except Exception:  # échec : retenu pour la session seulement, jamais sur disque
                _ROIC_CACHE[yahoo_ticker] = (np.nan, np.nan)
    return _ROIC_CACHE[yahoo_ticker]


def add_roic(res: pd.DataFrame, progress_cb=None, workers: int = 4) -> pd.DataFrame:
    """Ajoute les colonnes « ROIC % » (dernier exercice) et « ROIC moy % » (une requête double
    par titre, en parallèle). Appelé uniquement sur la short-list, après les autres filtres.
    Le ROIC n'a de sens que pour les actions : les ETF ne sont pas interrogés (NaN)."""
    tickers = res.loc[res["section"] == "Stock", "yahoo"].drop_duplicates().tolist()
    out = {}
    if tickers:
        with ThreadPoolExecutor(max_workers=workers) as ex:
            futs = {ex.submit(get_roic, y): y for y in tickers}
            for i, f in enumerate(as_completed(futs), 1):
                out[futs[f]] = f.result()
                if progress_cb:
                    progress_cb(i / len(tickers))
    if CACHE and _ROIC_NEW:
        CACHE.save_roic(dict(_ROIC_NEW))
        _ROIC_NEW.clear()
    res = res.copy()
    res["ROIC %"] = res["yahoo"].map(lambda y: out.get(y, (np.nan, np.nan))[0])
    res["ROIC moy %"] = res["yahoo"].map(lambda y: out.get(y, (np.nan, np.nan))[1])
    return res


def filter_roic(res: pd.DataFrame, roic_min: float, roic_col: str,
                keep_missing: bool) -> pd.DataFrame:
    """Filtre sur le ROIC minimum. S'applique aux actions uniquement : les ETF passent toujours."""
    roic = res[roic_col]
    ok = roic >= roic_min  # NaN -> False
    if keep_missing:
        ok = ok | roic.isna()
    ok = ok | (res["section"] != "Stock")
    return res[ok]

# Calculs

def compute_metrics(close: pd.Series, window_start: pd.Timestamp,
                    min_points: int, last_day: pd.Timestamp) -> dict | None:
    s = close.dropna()
    if s.empty or s.index[-1] < last_day - pd.Timedelta(days=7):
        return None  # cotation arrêtée ou données obsolètes
    w = s[s.index >= window_start]
    if len(w) < max(min_points, 5):
        return None  # historique trop court sur la fenêtre
    dd = w / w.cummax() - 1.0
    last, high = w.iloc[-1], w.max()
    ma200 = s.rolling(200).mean().iloc[-1] if len(s) >= 200 else np.nan
    return {
        "Dernier cours": last,
        "Plus haut période": high,
        "Date plus haut": w.idxmax().date(),
        "DD actuel %": (last / high - 1) * 100,
        "DD max %": dd.min() * 100,
        "Date creux": dd.idxmin().date(),
        "Perf période %": (last / w.iloc[0] - 1) * 100,
        "vs MM200 %": (last / ma200 - 1) * 100 if pd.notna(ma200) else np.nan,
    }


def run_scan(universe: pd.DataFrame, window_start: pd.Timestamp, end: pd.Timestamp,
             dd_threshold: float, dd_column: str, above_ma200: bool,
             progress_cb=None, pe_max: float | None = None, pe_min: float = 0.0,
             pe_col: str = "PER", keep_missing_pe: bool = False,
             show_pe: bool = False, pe_progress_cb=None,
             roic_min: float | None = None, roic_col: str = "ROIC moy %",
             keep_missing_roic: bool = False, show_roic: bool = False,
             roic_progress_cb=None):
    """Retourne (résultats filtrés, prix, liste des tickers sans données, nb analysés)."""
    # On charge au moins ~14 mois pour pouvoir calculer la MM200
    fetch_start = min(window_start, end - pd.Timedelta(days=420))
    tickers = universe["yahoo"].tolist()
    prices = fetch_prices(
        tickers, fetch_start.strftime("%Y-%m-%d"),
        (end + pd.Timedelta(days=1)).strftime("%Y-%m-%d"), progress_cb,
    )
    if prices.empty:
        return pd.DataFrame(), prices, universe["ticker"].tolist(), 0

    expected = (end - window_start).days * 252 / 365
    min_points = int(0.7 * expected)
    last_day = prices.index.max()

    rows, missing = [], []
    for y in tickers:
        m = compute_metrics(prices[y], window_start, min_points, last_day) \
            if y in prices.columns else None
        if m is None:
            missing.append(y)
        else:
            rows.append({"yahoo": y, **m})

    info = universe[["yahoo", "ticker", "desc", "section", "currency", "market"]]
    missing_xtb = info.loc[info["yahoo"].isin(missing), "ticker"].tolist()
    if not rows:
        return pd.DataFrame(), prices, missing_xtb, 0

    res = pd.DataFrame(rows).merge(info, on="yahoo")
    analysed = len(res)
    res = res[res[dd_column] <= -dd_threshold]
    if above_ma200:
        res = res[res["vs MM200 %"] > 0]
    res = res.sort_values(dd_column).reset_index(drop=True)
    # PER : récupéré seulement pour les titres déjà retenus (une requête par titre)
    if not res.empty and (pe_max is not None or show_pe):
        res = add_pe(res, pe_progress_cb)
        if pe_max is not None:
            res = filter_pe(res, pe_min, pe_max, pe_col, keep_missing_pe)
            res = res.reset_index(drop=True)
    # ROIC : après le PER, donc seulement pour les titres qui ont survécu aux autres filtres
    if not res.empty and (roic_min is not None or show_roic):
        res = add_roic(res, roic_progress_cb)
        if roic_min is not None:
            res = filter_roic(res, roic_min, roic_col, keep_missing_roic)
            res = res.reset_index(drop=True)
    return res, prices, missing_xtb, analysed

def bdays(first: str, last: str) -> pd.DatetimeIndex:
    return pd.bdate_range(first, last)


class FakeMarket:
    """Réseau simulé : une série par ticker, tronquée à `today`, avec journal des appels."""

    def __init__(self, tickers, today="2026-02-20", dead=(), factor=None):
        self.tickers = set(tickers)
        self.dead = set(dead)
        self.today = pd.Timestamp(today)
        self.factor = factor or {}   # ticker -> facteur d'ajustement appliqué à tout l'historique
        self.calls: list[tuple] = []
        self.fail: set[str] = set()  # tickers qui « échouent » silencieusement

    def price(self, t, d):
        base = 100 + (sum(map(ord, t)) % 50)
        return (base + (d - pd.Timestamp("2026-01-01")).days * 0.1) * self.factor.get(t, 1.0)

    def __call__(self, tickers, start, end, cb=None):
        self.calls.append((tuple(tickers), start, end))
        out = {}
        for t in tickers:
            if t in self.dead or t in self.fail or t not in self.tickers:
                continue
            idx = bdays(start, min(pd.Timestamp(end) - pd.Timedelta(days=1), self.today))
            out[t] = pd.Series([self.price(t, d) for d in idx], index=idx)
        if cb:
            cb(1.0)
        return pd.DataFrame(out) if out else pd.DataFrame()


class Clock:
    def __init__(self, t="2026-02-20 10:00"):
        self.t = datetime.fromisoformat(t)

    def __call__(self):
        return self.t

    def advance(self, **kw):
        self.t += timedelta(**kw)


class CacheCase(unittest.TestCase):
    def setUp(self):
        self.dir = tempfile.TemporaryDirectory()
        self.addCleanup(self.dir.cleanup)
        self.clock = Clock()

def fake_download(alive: set[str], calls: list | None = None):
    def _dl(tickers, start, end):
        if calls is not None:
            calls.append(tuple(tickers))
        cols = {t: [1.0] for t in tickers if t in alive}
        return pd.DataFrame(cols, index=[pd.Timestamp.today().normalize()])
    return _dl


def universe(rows):
    """rows : (ticker XTB, desc) -> DataFrame comme celui de load_universe."""
    df = pd.DataFrame(rows, columns=["ticker", "desc"])
    df["yahoo"] = df["ticker"].map(ds.to_yahoo)
    return df

YEARS = pd.to_datetime(["2025-12-31", "2024-12-31", "2023-12-31", "2022-12-31"])  # ordre Yahoo : récent d'abord


def income(ebit, tax=None):
    rows = {"Operating Income": ebit}
    if tax is not None:
        rows["Tax Rate For Calcs"] = tax
    return pd.DataFrame(rows, index=YEARS[:len(ebit)]).T


def balance(**rows):
    n = len(next(iter(rows.values())))
    return pd.DataFrame(rows, index=YEARS[:n]).T


class FakeClock:
    def __init__(self, t="2026-02-20 10:00"):
        self.t = datetime.fromisoformat(t)

    def __call__(self):
        return self.t

    def advance(self, **kw):
        self.t += timedelta(**kw)


class TestComputeRoic(unittest.TestCase):
    def test_uses_invested_capital_and_tax_rate(self):
        inc = income([200.0, 100.0], tax=[0.25, 0.20])
        bal = balance(**{"Invested Capital": [1000.0, 800.0]})
        last, avg = ds.compute_roic(inc, bal)
        self.assertAlmostEqual(last, 200 * 0.75 / 1000 * 100)          # 15 %
        self.assertAlmostEqual(avg, (15.0 + 100 * 0.8 / 800 * 100) / 2)  # (15 + 10) / 2

    def test_fallback_invested_capital_from_debt_equity_cash(self):
        inc = income([300.0], tax=[0.0])
        bal = balance(**{"Total Debt": [500.0], "Stockholders Equity": [700.0],
                         "Cash And Cash Equivalents": [200.0]})
        last, _ = ds.compute_roic(inc, bal)
        self.assertAlmostEqual(last, 300 / 1000 * 100)

    def test_missing_tax_uses_default_and_absurd_tax_is_replaced(self):
        bal = balance(**{"Invested Capital": [1000.0]})
        a, _ = ds.compute_roic(income([100.0]), bal)
        b, _ = ds.compute_roic(income([100.0], tax=[0.95]), bal)      # taux aberrant -> repli
        expected = 100 * (1 - ds.ROIC_DEFAULT_TAX) / 1000 * 100
        self.assertAlmostEqual(a, expected)
        self.assertAlmostEqual(b, expected)

    def test_non_positive_invested_capital_is_ignored(self):
        inc = income([100.0, 100.0], tax=[0.0, 0.0])
        bal = balance(**{"Invested Capital": [500.0, -50.0]})
        last, avg = ds.compute_roic(inc, bal)
        self.assertAlmostEqual(last, 20.0)
        self.assertAlmostEqual(avg, 20.0)                              # l'exercice négatif est exclu

    def test_only_negative_capital_gives_nan(self):
        last, avg = ds.compute_roic(income([100.0], tax=[0.0]),
                                    balance(**{"Invested Capital": [-10.0]}))
        self.assertTrue(np.isnan(last) and np.isnan(avg))

    def test_missing_rows_give_nan(self):
        self.assertTrue(np.isnan(ds.compute_roic(pd.DataFrame(), pd.DataFrame())[0]))
        no_ebit = pd.DataFrame({"Net Income": [1.0]}, index=YEARS[:1]).T
        self.assertTrue(np.isnan(ds.compute_roic(no_ebit, balance(**{"Invested Capital": [1.0]}))[0]))
        self.assertTrue(np.isnan(ds.compute_roic(income([1.0]), balance(**{"Total Debt": [1.0]}))[0]))

    def test_last_is_most_recent_year_whatever_the_column_order(self):
        inc = income([300.0, 100.0, 100.0], tax=[0.0, 0.0, 0.0])
        bal = balance(**{"Invested Capital": [1000.0, 1000.0, 1000.0]})
        last, avg = ds.compute_roic(inc, bal)
        self.assertAlmostEqual(last, 30.0)
        self.assertAlmostEqual(avg, (30 + 10 + 10) / 3)
        # même résultat si Yahoo renvoie les colonnes dans l'autre sens
        last2, avg2 = ds.compute_roic(inc[inc.columns[::-1]], bal[bal.columns[::-1]])
        self.assertAlmostEqual(last2, last)
        self.assertAlmostEqual(avg2, avg)

    def test_average_is_limited_to_recent_years(self):
        n = ds.ROIC_MAX_YEARS + 2
        idx = pd.date_range(end="2025-12-31", periods=n, freq="YE")
        inc = pd.DataFrame({"Operating Income": [1000.0] + [100.0] * (n - 1)}, index=idx[::-1]).T
        bal = pd.DataFrame({"Invested Capital": [1000.0] * n}, index=idx[::-1]).T
        inc.loc["Tax Rate For Calcs"] = 0.0
        _, avg = ds.compute_roic(inc, bal)
        self.assertAlmostEqual(avg, (100 + 10 * (ds.ROIC_MAX_YEARS - 1)) / ds.ROIC_MAX_YEARS)


class TestGetRoicCache(unittest.TestCase):
    def setUp(self):
        self.dir = tempfile.TemporaryDirectory()
        self.addCleanup(self.dir.cleanup)
        self.addCleanup(ds.configure_cache, None)
        ds._ROIC_CACHE.clear()
        ds._ROIC_NEW.clear()
        self.addCleanup(ds._ROIC_CACHE.clear)
        self.addCleanup(ds._ROIC_NEW.clear)
        self.tk = mock.Mock(income_stmt=income([200.0], tax=[0.0]),
                            balance_sheet=balance(**{"Invested Capital": [1000.0]}))

    def test_reads_then_writes_disk_cache(self):
        ds.configure_cache(self.dir.name)
        with mock.patch.object(ds.yf, "Ticker", return_value=self.tk) as tk:
            self.assertEqual(ds.get_roic("A"), (20.0, 20.0))
            ds.add_roic(pd.DataFrame({"yahoo": ["A"], "section": ["Stock"]}))  # persiste
            ds._ROIC_CACHE.clear()                                             # nouvelle « session »
            self.assertEqual(ds.get_roic("A"), (20.0, 20.0))
            self.assertEqual(tk.call_count, 1)                                 # 2e lecture : disque
        self.assertTrue((Path(self.dir.name) / "roic.json").exists())

    def test_failed_request_is_never_persisted(self):
        ds.configure_cache(self.dir.name)
        with mock.patch.object(ds.yf, "Ticker", side_effect=ConnectionError("down")):
            r = ds.get_roic("A")
            ds.add_roic(pd.DataFrame({"yahoo": ["A"], "section": ["Stock"]}))
        self.assertTrue(np.isnan(r[0]))
        self.assertFalse((Path(self.dir.name) / "roic.json").exists())

    def test_empty_statements_are_not_persisted(self):
        ds.configure_cache(self.dir.name)
        empty = mock.Mock(income_stmt=pd.DataFrame(), balance_sheet=pd.DataFrame())
        with mock.patch.object(ds.yf, "Ticker", return_value=empty):
            self.assertTrue(np.isnan(ds.get_roic("A")[0]))
            ds.add_roic(pd.DataFrame({"yahoo": ["A"], "section": ["Stock"]}))
        self.assertFalse((Path(self.dir.name) / "roic.json").exists())

    def test_etf_is_not_queried(self):
        with mock.patch.object(ds.yf, "Ticker", return_value=self.tk) as tk:

        self.assertTrue(np.isnan(out.loc[out["yahoo"] == "E", "ROIC %"].iloc[0]))

    def test_add_roic_without_stocks_does_not_crash(self):

class TestDiskCacheRoic(unittest.TestCase):
    def setUp(self):
        self.dir = tempfile.TemporaryDirectory()
        self.addCleanup(self.dir.cleanup)
        self.clock = FakeClock()

    def mk(self, **kw):
        return cache_mod.DiskCache(self.dir.name, now=self.clock, **kw)

    def test_roundtrip_with_nan(self):
        self.mk().save_roic({"A": (12.5, 10.0), "B": (np.nan, np.nan)})
        got = self.mk().get_roic("A")
        self.assertEqual(got, (12.5, 10.0))
        b = self.mk().get_roic("B")
        self.assertTrue(np.isnan(b[0]) and np.isnan(b[1]))

    def test_expiry_unknown_and_refresh(self):
        c = self.mk(roic_ttl_hours=24)
        c.save_roic({"A": (12.5, 10.0)})
        self.assertIsNone(c.get_roic("ZZZ"))
        self.clock.advance(hours=23)
        self.assertEqual(self.mk(roic_ttl_hours=24).get_roic("A"), (12.5, 10.0))
        self.clock.advance(hours=2)
        self.assertIsNone(self.mk(roic_ttl_hours=24).get_roic("A"))
        self.assertIsNone(self.mk(refresh=True).get_roic("A"))

    def test_default_ttl_is_a_week(self):
        self.mk().save_roic({"A": (1.0, 1.0)})
        self.clock.advance(days=6)
        self.assertIsNotNone(self.mk().get_roic("A"))
        self.clock.advance(days=2)
        self.assertIsNone(self.mk().get_roic("A"))

    def test_corrupt_file_is_ignored(self):
        (Path(self.dir.name) / "roic.json").write_text("{pas du json", encoding="utf-8")
        self.assertIsNone(self.mk().get_roic("A"))

class TestCliRoic(unittest.TestCase):
    ROIC = {"AAA.PA": (8.0, 14.0), "BBB.PA": (30.0, 4.0)}

    def setUp(self):
        self.dir = tempfile.TemporaryDirectory()
        self.addCleanup(self.dir.cleanup)
        self.addCleanup(ds.configure_cache, None)
        ds._ROIC_CACHE.clear()
        ds._ROIC_NEW.clear()
        self.addCleanup(ds._ROIC_CACHE.clear)
        self.addCleanup(ds._ROIC_NEW.clear)
        self.csv = os.path.join(self.dir.name, "u.csv")
        pd.DataFrame({"ticker": ["AAA.FR", "BBB.FR"], "desc": ["Aaa", "Bbb"],
                      "section": ["Stock", "Stock"], "currency": ["EUR", "EUR"]}
                     ).to_csv(self.csv, index=False)
        today = pd.Timestamp.today().normalize()
        idx = pd.bdate_range(today - pd.Timedelta(days=500), today)
        n = len(idx)
        decline = np.concatenate([np.full(n - 40, 100.0), np.linspace(100, 40, 40)])
        self.frame = pd.DataFrame({"AAA.PA": decline, "BBB.PA": decline * 2}, index=idx)

    def run_cli(self, *args):
        def fake_chunk(tickers, start, end):
            cols = [t for t in tickers if t in self.frame.columns]
            return self.frame.loc[start:end, cols]
        out, err = io.StringIO(), io.StringIO()
        with mock.patch.object(sys, "argv", ["cli.py", *args, "--csv", self.csv]), \
                mock.patch.object(ds, "fetch_chunk", fake_chunk), \
                mock.patch.object(ds, "get_roic", lambda y: self.ROIC.get(y, (np.nan, np.nan))), \
                mock.patch.object(ds.time, "sleep", lambda s: None), \
                redirect_stdout(out), redirect_stderr(err):
            code = cli.main()
        return code, out.getvalue(), err.getvalue()

    def names(self, out):
        return [line.split()[0] for line in out.strip().splitlines()]

    def test_without_roic_flag_nothing_is_filtered(self):
        _, out, _ = self.run_cli("40", "3m", "--no-cache")
        self.assertEqual(self.names(out), ["AAA.FR", "BBB.FR"])

    def test_default_uses_average_roic(self):
        _, out, _ = self.run_cli("40", "3m", "--no-cache", "--roic-min", "10")
        self.assertEqual(self.names(out), ["AAA.FR"])          # moyenne 14 >= 10 ; BBB : moyenne 4

    def test_roic_type_last(self):
        _, out, _ = self.run_cli("40", "3m", "--no-cache", "--roic-min", "10", "--roic-type", "last")
        self.assertEqual(self.names(out), ["BBB.FR"])          # dernier exercice 30 >= 10 ; AAA : 8

    def test_nothing_matches(self):
        code, out, _ = self.run_cli("40", "3m", "--no-cache", "--roic-min", "50")
        self.assertEqual(code, 0)
        self.assertEqual(out.strip(), "")

    def test_show_roic_adds_columns_without_filtering(self):
        _, out, _ = self.run_cli("40", "3m", "--no-cache", "--show-roic", "-v")
        self.assertIn("ROIC %", out)
        self.assertIn("ROIC moy %", out)
        self.assertIn("AAA.FR", out)
        self.assertIn("BBB.FR", out)

    def test_keep_no_roic(self):
        ds_roic = {"AAA.PA": (8.0, 14.0)}                      # BBB sans ROIC
        with mock.patch.object(type(self), "ROIC", ds_roic):
            _, out, _ = self.run_cli("40", "3m", "--no-cache", "--roic-min", "10")
            self.assertEqual(self.names(out), ["AAA.FR"])
            _, out, _ = self.run_cli("40", "3m", "--no-cache", "--roic-min", "10", "--keep-no-roic")
            self.assertEqual(self.names(out), ["AAA.FR", "BBB.FR"])
